🔹 Basic C++ Syntax

Hello World

#include <iostream>
using namespace std;

int main() {
    cout << "Hello World";
    return 0;
}

Input & Output

int age;

cin >> age;
cout << age;

🔹 Conditions

If-Else

if (condition) {
    // code
}
else {
    // code
}

Switch

switch(choice) {

    case 1:
        cout << "One";
        break;

    case 2:
        cout << "Two";
        break;

    default:
        cout << "Invalid";
}

🔹 Loops

For Loop

for(int i = 0; i < 5; i++) {
    cout << i;
}

While Loop

while(condition) {
    // code
}

🔹 Functions

int add(int a, int b) {
    return a + b;
}

int main() {
    cout << add(2, 3);
}
💡 Function = reusable block of code.

🔹 Arrays

Declare & Access

int arr[5] = {10, 20, 30, 40, 50};

cout << arr[0];

Traverse Array

for(int i = 0; i < 5; i++) {
    cout << arr[i] << " ";
}

🔹 Strings

string name = "Sakshi";

cout << name;
cout << name.length();

🔹 Useful STL

Vector

#include <vector>

vector<int> v;

v.push_back(10);
v.push_back(20);

v.pop_back();

Sort

#include <algorithm>

sort(v.begin(), v.end());

Reverse

reverse(v.begin(), v.end());
